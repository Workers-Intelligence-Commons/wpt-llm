# The provider stack. This is the whole point of the module: these dependencies
# live in the loaded `cdylib` instead of in every host binary.
# `whatsapp-web` is here because `StartChannel` deliberately *permits* the
# WhatsApp Web configuration — it is the one WhatsApp variant with a real listen
# loop. Without the feature the factory compiles its gate-off branch, returns no
# provider, and the request fails as `NotConfigured`: a guard that allows a
# configuration the build cannot serve.
tinychannels = { path = "../..", features = ["email", "lark", "whatsapp-web", "relay-websocket"] }
# The transport-free vocabulary shared with the host.
tinychannels-bus = { path = "../tinychannels-bus" }
# TinyBus provides the typed service interface and dynamic module host ABI.
tinybus = { version = ">=0.1.4, <0.2", default-features = false, features = ["macros", "modules"] }
# The module-side SDK owns its runtime and exports the stable C entrypoints.
tinybus-module = ">=0.1.4, <0.2"
# Provider listen loops are long-lived tasks the module owns.
tokio = { version = "1", features = ["macros", "rt-multi-thread", "sync", "time"] }
# Storage over the bus: the module reaches host storage through the same port
# traits a linked library gets, never through per-module persistence callbacks.
